import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Real-Time Messaging" icon="rocket">
    WebSocket connectivity with automatic reconnection and exponential backoff. Messages appear instantly as they're posted.
  </Card>
  <Card title="Threaded Conversations" icon="comment">
    Open a side panel to view and reply to threaded discussions without leaving the main channel view.
  </Card>
  <Card title="Markdown Rendering" icon="document">
    Posts rendered with Glamour for rich terminal output — code blocks, links, emphasis, and more.
  </Card>
  <Card title="Command Palette" icon="magnifier">
    Fuzzy-filter overlay for slash commands. Open with Ctrl+K and start typing to find commands instantly.
  </Card>
  <Card title="Post Search" icon="magnifier">
    Search posts in the current channel with Ctrl+S. Navigate results and jump to matching messages.
  </Card>
  <Card title="Customizable Themes" icon="sun">
    Four built-in themes — Tokyo Night, Catppuccin, Kanagawa, and Nightfox — plus JSON-based custom theme support.
  </Card>
  <Card title="Unread Badges" icon="information">
    Per-channel unread message counts displayed in the sidebar so you always know where new messages are.
  </Card>
  <Card title="Post Tagging" icon="list-format">
    Tag posts with `#tag` syntax or the tag picker overlay. Search by tags to find relevant conversations quickly.
  </Card>
  <Card title="Keyboard Driven" icon="laptop">
    Navigate entirely by keyboard. Tab between panes, toggle panels, open overlays — no mouse required.
  </Card>
</CardGrid>
